package caixaeletronico;

import java.util.ArrayList;

import java.util.List;      

public class CaixaEletronico implements ICaixaEletronico { // IMPLEMENTAÇÃO DA INTERFACE PRINCIPAL NA CLASSE

// CRIAÇÃO DAS MATRIZ DE CÉDULAS, IMPLEMENTAÇÃO DE QUANTIDADE (1 CAMPO MOSTRANDO O VALOR DA CÉDULA E 2 CAMPO MOSTRANDO A QUANTIDADE DA DEVIDA CÉDULA)
    private int[][] cedulas = {
        {100, 100}, {50, 200}, {20, 300}, {10, 350}, {5, 450}, {2, 500}
    };

// O CAIXA IRÁ PARAR DE FUNCIONAR CASO O TOTAL CHEGUE A 0
    private int cotaMinima = 0;
    
// ESSA OPERAÇÃO ARMAZENA TODAS AS AÇÕES, RESULTADOS E OPERAÇÕES QUE O BANCO TEVE
    private List<String> extrato = new ArrayList<>();

// USO DE OVERRIDE PARA NÃO TER ERRO DE NOMEAÇÃO DOS MÉTODOS E GERAR ERROS NO CÓDIGO (GARANTE A IGUALDADE DOS MÉTODOS) @Override
    public String pegaValorTotalDisponivel() {
        int total = 0;

// OCORRE A MULTIPLICAÇÃO DA MATRIZ QUE POSSUI OS VALORES E A QUANTIDADE DE CÉDULAS, DEPOIS DA MULTIPLICAÇÃO VIRÁ A SOMA E COM FIM DANDO O RESULTADO TOTAL QUE POSSUI NO BANCO

        for (int i = 0; i < cedulas.length; i++) {
            total += cedulas[i][0] * cedulas[i][1];
        }
        return "Total disponível: R$ " + total;
    }

    @Override
    public String reposicaoCedulas(Integer cedula, Integer quantidade) {
// ESSA OPERAÇÃO É O LOCAL ONDE OCORRE AS REPOSIÇÕES, COMEÇANDO PELA VERIFICAÇÃO DE QUAL CÉDULA SERÁ ESCOLHIDA E QUANTAS QUANTIDADES IRÃO PARA A DEVIDA CÉDULA
        for (int i = 0; i < cedulas.length; i++) {
            if (cedulas[i][0] == cedula) {
                cedulas[i][1] += quantidade;

// IRÁ APARECER NA TELA QUAL FOI A QUANTIDADE DE REPOSIÇÃO E QUAL CÉDULA FOI ESCOLHIDA
                extrato.add("REPOSIÇÃO: " + quantidade + " NOTAS DE R$ " + cedula);
                return "REPOSIÇÃO REALIZADA COM SUCESSO:";
            }
        }
// CASO NÃO POSITIVO RETORNA QUE FOI INVÁLIDO
        return "Cédula inválida";
    }

    @Override
// UTILIZAÇÃO DO STRINGBUILDER PARA OTIMIZAÇÃO E MENOS OCUPAÇÃO DE DADOS NA MEMÓRIA, OU SEJA ELE VAI DERECIONAR O TEXTO DIRETAMENTE PARA O APPEND
// UTILIZAÇÃO DE APPEND PARA MODIFICAR OBJETOS E ADICIONAR DADOS OU TEXTOS NO FINAL DO STRINGBUILDER (COLOCANDO OS VALORES - E O ESPAÇO)
     public String pegaRelatorioCedulas() {
        StringBuilder relatorio = new StringBuilder();
        for (int i = 0; i < cedulas.length; i++) {
            relatorio.append(cedulas[i][0]).append(" - ").append(cedulas[i][1]).append("\n");
        }
        return relatorio.toString();
    }

 @Override
    public String sacar(Integer valor) {
       // VERIFICAÇÃO SE O VALOR QUE QUEIRA SACAR É NEGATIVO (CASO FOR DARÁ INVÁLIDO)
    	if (valor <= 0) {
            return "Valor inválido";
        }
    	
    
    	int totalAtual = 0;
        for (int i = 0; i < cedulas.length; i++) {
            totalAtual += cedulas[i][0] * cedulas[i][1];
        }
// VERIFICAÇÃO SE O VALOR DO SAQUE É MENOR QUE A COTA MÍNIMA
        if ((totalAtual - valor) < cotaMinima) {
            return "Caixa Vazio: Chame o Operador"; 
        }
// VERIFICAÇÃO CASO O USUSÁRIO COLOQUE UM VALOR MAIOR DO QUE POSSUI NO TOTAL DO BANCO, ASSIM IMPOSSIBILITANDO O SAQUE
        if (valor > totalAtual) {
            return "Saldo insuficiente";
        }

        int restante = valor; // OPREAÇÃO QUE VAI REGISTRAR O QUANTO DE VALOR FALTA PARA O USUÁRIO
        int totalCedulasSaque = 0; // TOTAL DE CÉDULAS QUE FOI RETIRADA DO SAQUE
        int[] quantidadePorNota = new int[cedulas.length]; // ARRAY QUE VAI CALCULAR QUANTAS NOTAS SERÁ ENTREGUE AO USUÁRIO

// OPERAÇÃO DE CONTROLE PARA NO SAQUE COLOCAR OS MAIORES VALORES DE NOTAS PRIMEIRO (DAR O MÁXIMO DE QUANTIDADE QUE CONSEGUIR DE 100 E 50)
        for (int i = 0; i < cedulas.length; i++) {
            int valorNota = cedulas[i][0];
            int disponivel = cedulas[i][1];
// O MATH MIN VAI ANALISAR SE A QUANTIDADE PEGA É DO ESTOQUE OU DO QUE PRECISA (ÁNALISE DE VALORES DO ESTOQUE)
            int qtdusar = Math.min(restante / valorNota, disponivel); 

            quantidadePorNota[i] = qtdusar; // GUARDA NO ARRAY QUANTAS NOTAS SERÃO UTILIZADAS
            restante -= qtdusar * valorNota; // CONTA QUANTAS NOTAS AINDA FALTAM PARA COBRIR O VALOR DO USUÁRIO
            totalCedulasSaque += qtdusar; // CONTA QUANTAS CÉDULAS FORAM ENTREGUES
        }

// SE O RESTANTE DE CÉDULAS FOR IGUAL A 0, OU SEJA NÃO POSSUI MAIS CÉDULAS NO BANCO NÃO POSSIBILITANDO O SAQUE
        if (restante != 0) {
            return "Saque não realizado por falta de cédulas"; 
        }

     //APLICAÇÃO DO LIMITE DE 30 CÉDULAS PARA CADA SAQUE
        if (totalCedulasSaque > 30) {
            return "Limite de cédulas excedido";
        }

//ATUALIZAÇÃO DAS QUANTIDADES DE NOTAS QUE ESTÃO NO ESTOQUE DEPOIS DE UM SAQUE REALIZADO
        for (int i = 0; i < cedulas.length; i++) {
            cedulas[i][1] -= quantidadePorNota[i]; 
        }

        
        extrato.add("SAQUE: R$ " + valor + " E Saldo restante: R$ " + (totalAtual - valor));
        return "Saque realizado com sucesso";
    }

    // DEFINIÇÃO E VALIDANDO A COTA MÍNIMA
    public String armazenaCotaMinima(Integer minimo) {
        this.cotaMinima = minimo;
        return "Cota mínima definida";
    }

    // OPERAÇÃO QUE ESTABELECE A VISUALIZAÇÃO DO EXTRATO DO SAQUE E O SALDO RESTANTE DO TOTAL
    public void mostrarExtratoFinal() {
        System.out.println("EXTRATO DO SAQUE E ATUALIZAÇÃO DO SALDO RESTANTE:");
        for (String linha : extrato) {
            System.out.println(linha);
        }
    }

// OPERAÇÃO QUE SE CONECTA COM A INTERFACE GUI PARA ELA SER INSTACIADA ("PARA ELA SER INICIADA")
    public static void main(String[] args) {
        GUI janela = new GUI(CaixaEletronico.class);
        janela.show();
    }
}
